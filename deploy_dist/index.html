📖 วิธีตั้งค่า Google Sheets (คลิกดู)
  1. เปิด sheets.new สร้าง Google Sheet ใหม่
  2. เมนู Extensions → Apps Script
  3. ลบโค้ดเดิม วางโค้ดด้านล่างนี้ทั้งหมด แล้วกด Save (Ctrl+S)
  4. กด Deploy → New deployment
  5. เลือก type = Web app
  6. "Who has access" = Anyone → กด Deploy
  7. คัดลอก Web app URL มาวางในช่องด้านบน
  8. กด "Test Connect" เพื่อทดสอบ

// วางโค้ดนี้ใน Apps Script function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const data = JSON.parse(e.postData.contents); if (data.action === 'ping') { return ContentService.createTextOutput(JSON.stringify({ok:true,msg:'Connected'})) .setMimeType(ContentService.MimeType.JSON); } let sheet = ss.getSheetByName('Inspection'); if (!sheet) { sheet = ss.insertSheet('Inspection'); sheet.appendRow(['ID','Date','Code ID','Location','Area','Technician', 'PPE','Ground','Continuity','Voltage','Equipment', 'LocationOK','Result','Issues','Training','PhotoLinks','Note']); sheet.getRange(1,1,1,17).setFontWeight('bold') .setBackground('#1F4E79').setFontColor('#fff'); } // Upload photos to Drive const photoLinks = []; if (data.photos) { const folder = getOrCreateFolder('SAFETY-EE Photos'); Object.entries(data.photos).forEach(([key, b64]) => { if (!b64) return; const blob = Utilities.newBlob( Utilities.base64Decode(b64.split(',')[1]), 'image/jpeg', `${data.id}_${key}.jpg`); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); photoLinks.push(`${key}: ${file.getUrl()}`); }); } sheet.appendRow([ data.id, data.date, data.code_id, data.location_name, data.location_area, data.technician, data.ppe.join(','), data.ground, data.continuity, data.voltage, data.equipment.join(','), data.location_ok ? 'YES' : 'NO', data.result, data.issues.join(' | '), data.training.map(t=>t.issue+'→'+t.course).join(' | '), photoLinks.join('\n'), data.note||'' ]); return ContentService.createTextOutput(JSON.stringify({ok:true,id:data.id})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ok:false,err:err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function getOrCreateFolder(name) { const folders = DriveApp.getFoldersByName(name); return folders.hasNext() ? folders.next() : DriveApp.createFolder(name); } function doGet() { return ContentService.createTextOutput('SAFETY-EE Endpoint Active'); }
