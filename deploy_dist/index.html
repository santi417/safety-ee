📤 Google Sheets Sync
วาง URL ที่ได้จาก Apps Script Deploy → Web App
📖 วิธีตั้งค่า Google Sheets (คลิกดู)
  1. เปิด sheets.new สร้าง Google Sheet ใหม่
  2. เมนู Extensions → Apps Script
  3. ลบโค้ดเดิม วางโค้ดด้านล่างนี้ทั้งหมด แล้วกด Save (Ctrl+S)
  4. กด Deploy → New deployment
  5. เลือก type = Web app
  6. "Who has access" = Anyone → กด Deploy
  7. คัดลอก Web app URL มาวางในช่องด้านบน
  8. กด "Test Connect" เพื่อทดสอบ
📋 Apps Script Code:
// SAFETY-EE Apps Script v2 — Sheets + Drive + Email Alert const ALERT_EMAIL = 'santi@planbmedia.co.th'; // เปลี่ยนได้ const ALERT_ON = ['CRITICAL']; // เพิ่ม 'STOP','FAIL' ได้ function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const data = JSON.parse(e.postData.contents); if (data.action === 'ping') { return ContentService.createTextOutput(JSON.stringify({ok:true,msg:'Connected'})) .setMimeType(ContentService.MimeType.JSON); } let sheet = ss.getSheetByName('Inspection'); if (!sheet) { sheet = ss.insertSheet('Inspection'); sheet.appendRow(['ID','Date','Code ID','Location','Area','Technician', 'PPE','Ground','Continuity','Voltage','Equipment', 'LocationOK','Result','Issues','Training','PhotoLinks','Note']); sheet.getRange(1,1,1,17).setFontWeight('bold') .setBackground('#1F4E79').setFontColor('#fff'); } // Upload photos to Drive const photoLinks = []; const photoUrls = {}; if (data.photos) { const folder = getOrCreateFolder('SAFETY-EE Photos'); Object.entries(data.photos).forEach(([key, b64]) => { if (!b64) return; const blob = Utilities.newBlob( Utilities.base64Decode(b64.split(',')[1]), 'image/jpeg', `${data.id}_${key}.jpg`); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); photoLinks.push(`${key}: ${file.getUrl()}`); photoUrls[key] = file.getUrl(); }); } sheet.appendRow([ data.id, data.date, data.code_id, data.location_name, data.location_area, data.technician, data.ppe.join(','), data.ground, data.continuity, data.voltage, data.equipment.join(','), data.location_ok ? 'YES' : 'NO', data.result, data.issues.join(' | '), data.training.map(t=>t.issue+'→'+t.course).join(' | '), photoLinks.join('\n'), data.note||'' ]); // Email alert if (ALERT_ON.indexOf(data.result) !== -1) { sendAlertEmail(data, photoUrls); } return ContentService.createTextOutput(JSON.stringify({ok:true,id:data.id})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ok:false,err:err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function sendAlertEmail(data, photoUrls) { const color = data.result === 'CRITICAL' ? '#c00' : data.result === 'STOP' ? '#e06c00' : '#d32f2f'; const icon = data.result === 'CRITICAL' ? '🚨' : data.result === 'STOP' ? '🛑' : '❌'; const photoHtml = Object.entries(photoUrls).map(([k,u]) => `📷 ${k.replace('photo_','')}`).join(''); const issuesHtml = data.issues.map(i => `
  • ${i}
    • `).join(''); const html = `
    ${icon} ${data.result}
    SAFETY-EE Alert
    วันที่: ${data.date}
    Code ID: ${data.code_id}
    Location: ${data.location_name||'—'}
    พื้นที่: ${data.location_area||'—'}
    ช่าง: ${data.technician}
    Ground: ${data.ground} Ω
    Continuity: ${data.continuity} Ω
    Voltage: ${data.voltage} V
    ⚠️ ปัญหาที่พบ
      ${issuesHtml}
    ${photoHtml ? `
    📷 รูปหลักฐาน
    ${photoHtml}
    ` : ''}
    ⚡ Action Required: กรุณาตรวจสอบและแก้ไขโดยด่วน
    SAFETY-EE SYSTEM | Plan B Media Standard
    `; MailApp.sendEmail({ to: ALERT_EMAIL, subject: `${icon} [${data.result}] ${data.code_id} ${data.location_name||''} — ${data.technician}`, htmlBody: html }); } function getOrCreateFolder(name) { const folders = DriveApp.getFoldersByName(name); return folders.hasNext() ? folders.next() : DriveApp.createFolder(name); } function doGet() { return ContentService.createTextOutput('SAFETY-EE Endpoint Active v2'); }
